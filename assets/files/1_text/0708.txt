import time
#decorator로 사용되면 func가 호출하는 함수가 됩니다.
def deco(func):
    n = 0
    def inner():
        nonlocal n
        n = n + 1
        print(time.time())
        print(n)
        func()
    return inner

#target을 호출할 때 deco라는 함수가 리턴하는 함수를 호출
@deco
def target():
    print("running target")

target()
target()

#비니지스 로직을 가진 함수 - 팩토리얼 구하는 함수를 재귀(함수 내부에서 자기 자신을 다시 호출하는 것 - 코드는 단순해져서 읽기 쉽지만 메모리를 많이 사용)를 이용해서 구현

import time
#데코레이터를 위한 함수
def logging(func):
    def logged(*args):
        start = time.time()

        #실제 함수를 호출
        result = func(*args)
        print("함수 호출 결과:", result)

        end = time.time()
        elapsed = end - start
        print("함수가 작업에 걸린 시간:", elapsed)
        return result
    return logged


#팩토리얼 구하는 함수를 재귀를 이용해서 구현
@logging
def factorial(n):
    return 1 if n < 2 else n * factorial(n-1)

print(factorial(5))

=>Operator Overloading(연산자 오버로딩)
- 연산자의 기능을 변경해서 사용하는 것
- 예를 들면 +는 숫자 데이터끼리의 연산에 사용되는 것인데 str 클래스에서는 문자열을 결합하는 기능을 수행합니다.
- 연산자에 해당하는 __메소드이름__ 메서드를 클래스에서 다시 정의하면 연산자가 정의한 기능을 수행합니다.
==은 __eq__ 메서드의 결과를 리턴합니다.
인스턴스 이름을 print에 대입했을 때 나오는 문자열은 __str__ 메서드를 다시 정의한 것입니다.

#인스턴스 별로 번호 와 이름을 속성으로 갖는 클래스
class Student:
    def __init__(self, name, age):
        self.__name = name
        self.__age = age #앞에 __가 붙으면 속성이 private 이 됩니다. 클래스 외부에서 사용 불가

    @property
    def name(self):
        return self.__name
    @name.setter
    def name(self, name):
        print("name의 setter 호출")
        self.__name = name

    @property
    def age(self):
        return self.__age
    @age.setter
    def age(self, age):
        self.__age = age

    #+ 연산자의 기능을 overloading: __name을 결합
    def __add__(self, other):
        return self.__name + "," + other.__name

    #== 연산자의 기능을 overloading
    def __eq__(self, other):
        return len(self.__name) == len(other.__name)

    #인스턴스 이름만 기재했을 때 리턴되는 문자열을 만들어주는 함수
    def __str__(self):
        return self.__name
        
s1 = Student("adam", 30)
s2 = Student("rusia", 27)
s3 = Student("adam", 27)
print(s1 + s2) #연산자 오버로딩 때문에 name을 결합한 결과
print(s1)

2)Inheritance(상속)
=>개요
- 상속이란 하위 클래스가 상위 클래스의 모든 것을 물려받는 것
- 상위 클래스를 Based Class, Super Class 라고 하고 하위 클래스를 Derived Class, Sub Class 라고 합니다.
- 하나의 클래스로부터 상속받는 것을 단일 상속이라고 하고 여러 클래스로부터 상속받는 것을 다중 상속이라고 합니다.

=>상속의 목적
- 여러 클래스에 공통으로 존재하는 코드를 상위 클래스에 묶어서 중복을 제거하기 위해서
- 제공되는 클래스의 기능이 부족해서 기능을 확장하거나 추가하기 위해서

=>상속 방법
class 클래스이름(상위 클래스 이름):

=>상위 클래스의 메서드 호출
class Person:
    def superMethod(self):
        print("상위 클래스의 메서드")

#Person 클래스가 상위 클래스가 되도록 설정
class Student(Person):
    def subMethod(self):
        print("하위 클래스의 메서드")

s = Student()

=>상위 클래스에서 만든 인스턴스 변수 호출
- 상위 클래스에서 만든 인스턴스 변수를 하위 클래스의 인스턴스가 사용하고자 할 때는 __init__ 함수를 만들어서 상위 클래스의 __init__를 호출해야 합니다.
호출할 때는 super().__init__()로 호출합니다.

class Person:
    def __init__(self, num):
        self.num = num

    def superMethod(self):
        print("상위 클래스의 메서드")

#Person 클래스가 상위 클래스가 되도록 설정
class Student(Person):
    #상위 클래스에 매개변수가 있는 __init__이 있는 경우는 직접 호출해주어야 합니다.
    def __init__(self):
        super().__init__(1)

    def subMethod(self):
        print("하위 클래스의 메서드")

s = Student()
s.subMethod()
s.superMethod() #상위 클래스의 메서드 호출
print(s.num) #상위 클래스의 __init__ 에 매개변수가 있고 초기값이 없으면 이 구문은 에러
#이런 경우에는 하위 클래스에 __init__를 만들어서 super().__init__()를 호출해 주어야 합니다.

=>Method Overriding(메서드 재정의)
- 상위 클래스에 존재하는 메서드와 동일한 모양의 메서드를 하위 클래스에 만드는 것
- 동일한 메서드가 존재하는 경우에는 하위 클래스의 메서드가 호출됩니다.
- 기능 확장이 목적
- 기능 확장이 목적이므로 상위 클래스의 메서드를 호출하는 것이 일반적입니다.(특정한 프레임워크에서는 상위 클래스의 메서드를 호출하지 않으면 에러가 발생)
- 상위 클래스의 메서드 호출 구문은 정리나 파괴를 할 때는 뒤에서 호출하고 그 이외의 경우는 먼저 호출합니다.
하위 클래스의 인스턴스를 생성할 때 상위 클래스의 인스턴스를 먼저 생성하고 하위 클래스의 인스턴스를 생성하기 때문입니다.

class Person:
    def method(self):
        print("상위 클래스의 메서드")

#Person 클래스가 상위 클래스가 되도록 설정
class Student(Person):
    #메서드 오버라이딩: 기능 확장이 목적이므로 상위 클래스의 메서드를 호출해야 합니다.
    def method(self):
        super().method()
        print("하위 클래스의 메서드")
s = Student()
s.method()

- Abstract Method

import abc
#추상 클래스 - 인스턴스를 생성할 수 없습니다.
class Sample(metaclass = abc.ABCMeta):
    #추상 메서드
    @abc.abstractmethod
    def method(self):
        pass

#추상 클래스를 상속받으면 반드시 추상 메서드를 구현해야 합니다.
class Sub(Sample):
    #추상 메서드 구현
    def method(self):
        print("추상 메서드 구현")

#obj = Sample() #추상 클래스는 인스턴스를 만들 수 없기 때문에 에러
obj = Sub()
